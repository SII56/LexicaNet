const API_URL = "https://api.deepseek.com/chat/completions"; const MODEL = "deepseek-chat"; const SYSTEM_PROMPT = ` Generate an English lexical activation note for the target word. Learner: CEFR B2, productive vocabulary. Goal: help recall the word through semantic associations, not synonyms only. Return valid JSON only: {"word":"","definition":"","core":"","associations":[["","",""],["","",""],["","",""],["","",""]]} Rules: - word: target word only - definition: English, one sense, <=15 words - core: intuitive mental tag, 2-4 words - associations: 3-4 lines, 3-4 items each - mix common verbs/nouns/adjectives/concepts/opposites/situations - use common B1-B2 words; avoid rare/literary words - order from strongest recall cues to weaker cues - no explanation, no markdown The last association line should contain 3 common collocations or typical usage patterns if available. `.trim(); chrome.runtime.onInstalled.addListener(() => { chrome.contextMenus.create({ id: "generate-anki-tsv", title: "Generate Anki TSV: \"%s\"", contexts: ["selection"] }); }); chrome.contextMenus.onClicked.addListener(async (info) => { if (info.menuItemId !== "generate-anki-tsv") return; const word = (info.selectionText || "").trim(); if (!word) { console.error("No selected word."); return; } try { const { apiKey } = await chrome.storage.local.get(["apiKey"]); if (!apiKey) { throw new Error("No API key. Open extension options and save your DeepSeek API key."); } const data = await callDeepSeek(apiKey, word); const tsv = jsonToTsv(data); await copyToClipboard(tsv); console.log("Copied TSV:", tsv); } catch (err) { console.error(err); } }); async function callDeepSeek(apiKey, word) { const res = await fetch(API_URL, { method: "POST", headers: { "Content-Type": "application/json", "Authorization": `Bearer ${apiKey}` }, body: JSON.stringify({ model: MODEL, response_format: { type: "json_object" }, messages: [ { role: "system", content: SYSTEM_PROMPT }, { role: "user", content: `Target word: ${word}` } ], temperature: 0.4, max_tokens: 300 }) }); if (!res.ok) { throw new Error(`API error ${res.status}: ${await res.text()}`); } const body = await res.json(); const content = body.choices[0].message.content.trim(); return JSON.parse(content); } function jsonToTsv(data) { const assoc = data.associations .map(line => line.join(" · ")) .join("\\n"); return [ data.word, data.definition, data.core, assoc ].join("\t"); } async function copyToClipboard(text) { await navigator.clipboard.writeText(text); }
